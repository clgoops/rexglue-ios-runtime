cmake_minimum_required(VERSION 3.25)
project(rex_native_renderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(rex-native-frame STATIC src/frame.cpp src/upload_probe.cpp)
target_include_directories(rex-native-frame PUBLIC include)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(REX_NATIVE_SPIRV_CROSS_DIR "" CACHE PATH "Pinned SPIRV-Cross source directory (see DEPENDENCIES.md)")
if(REX_NATIVE_SPIRV_CROSS_DIR)
  if(TARGET spirv-cross-msl)
    # The SDK already uses the same pinned translator. Reuse its actual source.
    get_target_property(REX_NATIVE_SPIRV_CROSS_DIR spirv-cross-msl SOURCE_DIR)
  endif()
  include(cmake/readonly-storage.cmake)
  foreach(feature CLI ENABLE_TESTS ENABLE_HLSL ENABLE_CPP ENABLE_REFLECT ENABLE_C_API ENABLE_UTIL)
    set(SPIRV_CROSS_${feature} OFF CACHE BOOL "" FORCE)
  endforeach()
  set(SPIRV_CROSS_SKIP_INSTALL ON CACHE BOOL "" FORCE)
  if(NOT TARGET spirv-cross-msl)
    add_subdirectory("${REX_NATIVE_SPIRV_CROSS_DIR}" spirv-cross EXCLUDE_FROM_ALL)
  endif()
  add_library(rex-native-spirv STATIC src/spirv.cpp)
  target_include_directories(rex-native-spirv PUBLIC include)
  target_link_libraries(rex-native-spirv PUBLIC rex-native-frame PRIVATE spirv-cross-msl)
  add_executable(native-shader-export tools/shader-export.cpp)
  target_link_libraries(native-shader-export PRIVATE rex-native-spirv)
endif()
option(REX_NATIVE_BUILD_TESTS "Build native renderer contract tests" ON)
set(REX_NATIVE_REXSDK_DIR "" CACHE PATH "Pinned ReXGlue SDK for frontend adapters and real shader tests")
if(REX_NATIVE_REXSDK_DIR AND TARGET rex-native-spirv)
  add_library(rex-native-rex-adapters STATIC src/rex_constants.cpp src/rex_shared_memory.cpp src/rex_shader.cpp src/rex_draw_state.cpp src/rex_targets.cpp src/rex_formats.cpp src/rex_transfer_shader.cpp src/rex_target_transfers.cpp src/rex_textures.cpp src/rex_texture_formats.cpp src/rex_texture_layout.cpp src/rex_texture_kernels.cpp src/rex_primitives.cpp src/rex_state.cpp src/rex_resolve.cpp src/rex_target_resolve.cpp src/rex_dump_shader.cpp src/rex_resolve_kernels.cpp)
  target_include_directories(rex-native-rex-adapters PUBLIC include
    "${REX_NATIVE_REXSDK_DIR}/include" "${REX_NATIVE_REXSDK_DIR}/thirdparty/fmt/include"
    "${REX_NATIVE_REXSDK_DIR}/thirdparty/glslang"
    "${REX_NATIVE_REXSDK_DIR}/thirdparty/vulkan-headers/include"
    "${REX_NATIVE_REXSDK_DIR}/thirdparty/renderdoc"
    "${REX_NATIVE_REXSDK_DIR}/thirdparty/xxhash")
  target_link_libraries(rex-native-rex-adapters PUBLIC rex-native-spirv)
  target_include_directories(rex-native-rex-adapters PRIVATE "${REX_NATIVE_REXSDK_DIR}/src/graphics/shaders/vulkan_spirv")
  add_library(rex-native-ui STATIC src/rex_immediate.cpp)
  target_link_libraries(rex-native-ui PUBLIC rex-native-rex-adapters)
  add_library(rex-native-live STATIC src/rex_provider.cpp src/rex_command_processor.cpp)
  target_include_directories(rex-native-live PRIVATE "${REX_NATIVE_REXSDK_DIR}/thirdparty/simde" "${REX_NATIVE_REXSDK_DIR}/thirdparty/rapidjson/include" "${REX_NATIVE_REXSDK_DIR}/thirdparty/spdlog/include")
  target_compile_definitions(rex-native-live PRIVATE SPDLOG_FMT_EXTERNAL)
  target_link_libraries(rex-native-live PUBLIC rex-native-rex-adapters rex-native-ui)
endif()
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 23)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  set(REX_NATIVE_METAL_LIBRARY_TYPE SHARED CACHE STRING "Native Metal library linkage")
  add_library(rex-native-metal ${REX_NATIVE_METAL_LIBRARY_TYPE} src/metal_backend.mm)
  target_include_directories(rex-native-metal PUBLIC include)
  target_compile_options(rex-native-metal PRIVATE -fobjc-arc)
  target_link_libraries(rex-native-metal PUBLIC rex-native-frame
      "-framework Foundation" "-framework Metal" "-framework QuartzCore")
endif()
if(REX_NATIVE_BUILD_TESTS)
  enable_testing()
  add_executable(native-frame-tests tests/frame.cpp)
  target_link_libraries(native-frame-tests PRIVATE rex-native-frame)
  add_test(NAME native-frame COMMAND native-frame-tests)
  if(TARGET rex-native-spirv)
    add_executable(native-spirv-tests tests/spirv.cpp)
    target_include_directories(native-spirv-tests PRIVATE "${REX_NATIVE_SPIRV_CROSS_DIR}")
    target_link_libraries(native-spirv-tests PRIVATE rex-native-spirv)
    add_test(NAME native-spirv COMMAND native-spirv-tests)
    if(REX_NATIVE_REXSDK_DIR)
      include(tests/rex-shaders.cmake)
      add_executable(native-ui-tests tests/ui.cpp "${REX_NATIVE_REXSDK_DIR}/src/ui/immediate_drawer.cpp" "${REX_NATIVE_REXSDK_DIR}/src/ui/graphics_util.cpp")
      target_link_libraries(native-ui-tests PRIVATE rex-native-ui)
      target_compile_definitions(native-ui-tests PRIVATE NDEBUG)
      add_test(NAME native-ui COMMAND native-ui-tests)
      if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
        target_compile_definitions(native-ui-tests PRIVATE REX_NATIVE_TEST_METAL=1)
        target_link_libraries(native-ui-tests PRIVATE rex-native-metal)
      endif()
    endif()
  endif()
  if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_executable(native-metal-tests tests/metal.mm)
    target_compile_options(native-metal-tests PRIVATE -fobjc-arc)
    target_link_libraries(native-metal-tests PRIVATE rex-native-metal)
    add_test(NAME native-metal COMMAND native-metal-tests)
    if(TARGET rex-native-spirv AND REX_NATIVE_REXSDK_DIR)
      add_executable(native-metal-spirv-tests tests/metal-spirv.cpp)
      target_include_directories(native-metal-spirv-tests PRIVATE "${REX_NATIVE_REXSDK_DIR}/src/graphics/shaders/vulkan_spirv")
      target_link_libraries(native-metal-spirv-tests PRIVATE rex-native-metal rex-native-spirv)
      add_test(NAME native-metal-spirv COMMAND native-metal-spirv-tests)
    endif()
  endif()
endif()
